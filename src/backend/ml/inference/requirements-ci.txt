# Lightweight CI dependency set for the nightly GitHub Actions pipeline.
# Kept separate from the backend's runtime requirements.txt so the runner
# stays small — the API does not need pandas/lightgbm to serve static JSON.
# numpy 2.x breaks the lightgbm/scipy wheels of this era — pin to 1.x
# so the nightly seed job cannot die on an ABI mismatch.

pandas>=2.0,<3
scipy>=1.11
scikit-learn>=1.4
lightgbm>=4.0,<5
httpx>=0.27

# Spectral ingestion (v2 M-S1). CI-only on purpose: the serving runtime reads
# the resulting JSON and never imports either of these, which keeps the
# container cold start and its RAM ceiling untouched.
pystac-client>=0.8
rasterio>=1.3
planetary-computer>=1.0
